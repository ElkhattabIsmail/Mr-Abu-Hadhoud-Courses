void BinarySearch(int[] numbers, int target) {
    int left = 0, right = numbers.Length - 1;
    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (numbers[mid] == target) return;
        if (numbers[mid] < target) left = mid + 1;
        else right = mid - 1;
    }
