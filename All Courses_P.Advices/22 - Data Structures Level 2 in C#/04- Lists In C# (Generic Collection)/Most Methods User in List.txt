Example Usage
csharp
Copy
List<int> numbers = new List<int> { 1, 2, 3, 4, 5 };

// Adding elements
numbers.Add(6);
numbers.AddRange(new[] { 7, 8, 9 });

// Removing elements
numbers.Remove(3);
numbers.RemoveAt(0);

// Accessing elements
int first = numbers[0];
bool contains = numbers.Contains(5);

// Sorting and searching
numbers.Sort();
int index = numbers.BinarySearch(4);

// Iterating
numbers.ForEach(n => Console.WriteLine(n));

// Converting
int[] array = numbers.ToArray();
